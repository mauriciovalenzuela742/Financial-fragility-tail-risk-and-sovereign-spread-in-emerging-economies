\section{Pregunta}
\begin{frame}{La pregunta: dos riesgos domésticos de cola y el \textit{spread} soberano}
\begin{exampleblock}{\textit{Spread} soberano}
\footnotesize Prima que exige el mercado por el riesgo de incumplimiento de un gobierno emergente: rendimiento de su deuda en dólares menos el del Tesoro de EE.UU.\ (EMBI, en puntos básicos).
\end{exampleblock}
\vspace{-8pt}
\begin{columns}[T]
\column{0.48\textwidth}
\begin{block}{Riesgo financiero}
\footnotesize Fragilidad del sistema bancario: $\JLoss$ (Chari et al., 2024).
\end{block}
\column{0.48\textwidth}
\begin{block}{Riesgo económico}
\footnotesize Cola adversa del crecimiento: $\GaR$ (Adrian et al., 2019).
\end{block}
\end{columns}
\begin{alertblock}{Brecha y pregunta}
\footnotesize No se han cruzado \textbf{simultáneamente} los dos riesgos de cola domésticos, el financiero y el económico. ¿Cada uno encarece la deuda soberana, y su \textbf{coincidencia la amplifica}?
\end{alertblock}
\end{frame}

% 4
\begin{frame}{Hipótesis de trabajo}
\begin{enumerate}
\item \textbf{Cada riesgo importa.} La fragilidad bancaria y el riesgo de cola del crecimiento se asocian con un \textit{spread} mayor, y el riesgo económico sobrevive a las variables financieras.
\vspace{4pt}
\item \textbf{Amplificación.} Su coincidencia eleva el \textit{spread} más que la suma de sus efectos (si no, son \textbf{aditivos}).
\vspace{4pt}
\item \textbf{Resiliencia.} Con menos deuda, mejor balance fiscal y más reservas, ambos efectos son menores.
\vspace{4pt}
\item \textbf{Crisis.} En una crisis que afecta a todos, ese resguardo puede no bastar.
\end{enumerate}
\end{frame}

\begin{frame}{Metodología}
\begin{block}{Especificación principal (panel trimestral, 13 economías, 2004--2026, $N=765$)}
\small
\vspace{-6pt}
\[
\ln\EMBI_{i,t}=\alpha_i+\delta_t+\beta_1\ln\JLoss_{i,t-1}+\beta_2D_{i,t-1}+\beta_3\,(\ln\JLoss\times D)_{i,t-1}+\varepsilon_{i,t}
\]
\vspace{-8pt}
\end{block}
{\scriptsize \textbf{Donde:} $i$ país, $t$ trimestre; $\alpha_i$, $\delta_t$ efectos fijos de país y de tiempo; $D=-\GaR$ riesgo de cola; $\beta_1$, $\beta_2$ efecto de cada riesgo; $\beta_3$ interacción ($>0$: amplificación; $=0$: efectos aditivos).\par}
\vspace{4pt}
\small
\begin{itemize}
\item \textbf{Rezagos} de un trimestre, frente a la retroalimentación del \textit{spread}.
\item \textbf{Efectos fijos de tiempo}: absorben los factores financieros globales.
\item \textbf{Errores}: Driscoll--Kraay, \textit{cluster} por país y \textit{wild cluster bootstrap}.
\item \textbf{Extensiones}: resiliencia ($R$: deuda, balance fiscal, reservas) y crisis con y sin respaldo externo.
\end{itemize}
\end{frame}

% 10
\section{Conclusiones}
\begin{frame}{Conclusiones e implicancias}
\begin{enumerate}
\item El \textbf{riesgo económico} doméstico se asocia con el \textit{spread} soberano, más allá de las variables financieras.
\item Los dos riesgos son \textbf{aditivos}: no hay amplificación en promedio.
\item La \textbf{resiliencia} baja el \textit{spread} y protege en tiempos normales, no en crisis sistémicas.
\item La amplificación aparece \textbf{sin respaldo externo} y en economías de financiamiento externo: evidencia condicional.
\end{enumerate}
\vspace{4pt}
\begin{block}{Implicancia}
\small Monitorear juntos el riesgo financiero y el económico; el resguardo propio (deuda, balance fiscal, reservas) protege en tiempos normales, y en una crisis sistémica lo que protege es el respaldo externo.
\end{block}
{\scriptsize Pendiente: revisar las pruebas econométricas con el profesor de econometría; correlaciones, no causalidad.\par}
\end{frame}

\section{Anexos}

\begin{frame}{Anexo: Construcción de $\JLoss$: fragilidad bancaria sistémica}
\small
Pérdida conjunta esperada del sistema bancario ante un evento sistémico, en \% de la exposición (Chari et al., 2024).
\begin{enumerate}
\item \textbf{Probabilidad de incumplimiento de cada banco} (Merton--KMV): $D^*=L_{ST}+\tfrac12L_{LT}$,\quad $EDF=\Phi(-DD)$.
\item \textbf{Factor sistémico} país (Vasicek): $p_i(V)=\Phi\!\left(\frac{\Phi^{-1}(p_i)-\rho_iV}{\sqrt{1-\rho_i^2}}\right)$.
\item \textbf{Agregación por punto de silla} de la distribución de pérdidas del sistema.
\item \textbf{Métrica}: $\JLoss=\dfrac{EL+UL}{\sum_j EAD_j}\times100$.
\end{enumerate}
\vspace{2pt}
{\scriptsize \textbf{Donde:} $D^*$ punto de incumplimiento; $L_{ST}$, $L_{LT}$ pasivos de corto y largo plazo; $DD$ distancia al incumplimiento; $EDF$ frecuencia esperada de incumplimiento; $\Phi$ distribución normal estándar; $p_i$ PD del banco $i$; $V$ factor sistémico (retorno bursátil del país); $\rho_i$ correlación del banco con $V$; $EL$ pérdida esperada; $UL$ pérdida inesperada al 99\%; $EAD$ exposición; LGD $=45\%$. La cota de la malla es activa en el 98\% de los casos $\Rightarrow$ $\JLoss$ es \textbf{ordinal} y entra en logaritmos.\par}
\end{frame}

\begin{frame}{Anexo: Construcción del $\GaR$: riesgo de cola del crecimiento}
\begin{columns}[c]
\column{0.52\textwidth}
\footnotesize
\begin{enumerate}
\item \textbf{Índice de condiciones financieras} (FCI tipo CISS): tensión accionaria, de tasas y cambiaria, en términos reales (deflactadas por la inflación).
\item \textbf{Regresión cuantílica de panel}, un trimestre adelante:\\
{\footnotesize $Q(\Delta y_{i,t+1}\mid\cdot;\tau)=\alpha+\beta_1\Delta y_{i,t}+\beta_2FCI^{\perp}_{i,t}+\beta_3VIX_t+\mu_i$}
\item $\GaR_{i,t}$ = cuantil $\tau=5\%$; ventana expansiva (pseudo--tiempo real).
\end{enumerate}
\vspace{2pt}
{\scriptsize \textbf{Donde:} $Q(\cdot;\tau)$ cuantil $\tau$ del crecimiento; $\Delta y$ crecimiento del PIB; $FCI^{\perp}$ FCI sin el componente del VIX; $\mu_i$ efecto fijo de país. En la tesis $D\equiv-\GaR$: mayor $D$, cola más adversa. No es circular con el EMBI (corr.\ 0,985 sin el componente tipo \textit{spread}).\par}

\centering

{\scriptsize Chile: trimestre benigno vs.\ 2020Q2 ($\GaR=-13{,}4$ pp).}
\end{columns}
\end{frame}
