\section{Motivación y pregunta}

\begin{frame}{Motivación}
\begin{itemize}
\item En las crisis de las últimas dos décadas los \textit{spreads} soberanos emergentes se ampliaron más que los avanzados.
\item Dos fuentes candidatas:
  \begin{itemize}
  \item \textbf{Fragilidad bancaria doméstica}: eleva el pasivo contingente del fisco (rescates).
  \item \textbf{Riesgo a la baja del crecimiento}: deteriora la capacidad fiscal justo en los malos estados.
  \end{itemize}
\item La literatura las trata por separado:
  \begin{itemize}
  \item el nexo banca--soberano trata el crecimiento como fundamento de fondo;
  \item \textit{growth-at-risk} no llega a la prima soberana;
  \item Chari et al.\ (2024) interactúan la fragilidad con factores \emph{globales}, no con el ciclo doméstico.
  \end{itemize}
\end{itemize}
\vspace{4pt}
\begin{block}{Pregunta}
¿Amplifica la coincidencia de alta fragilidad bancaria y elevado riesgo de cola del crecimiento el \textit{spread} soberano por encima de la suma de sus efectos individuales?
\end{block}
\end{frame}

\begin{frame}{El mecanismo: un \textit{doom loop} banca--soberano}

\column{0.5\textwidth}
\centering

\end{tikzpicture}%
}
\column{0.5\textwidth}
\footnotesize
\textbf{1. Bucle banca--soberano} (Farhi--Tirole, 2018). Un choque deprime el precio de la deuda, erosiona el patrimonio bancario y fuerza un rescate financiado con más deuda: un multiplicador.

\vspace{4pt}
\textbf{2. $\JLoss$ como pasivo contingente} (\textit{bailout put}; Acharya--Drechsler--Schnabl, 2014). El mercado lo incorpora \textit{ex ante} en la prima.

\vspace{4pt}
\textbf{3. $\GaR$ como estado que activa el multiplicador} (Adrian--Boyarchenko--Giannone, 2019).

\vspace{4pt}
\textcolor{tesisred}{La flecha discontinua es también el principal problema de identificación.}
\end{columns}
\end{frame}

\begin{frame}{Hipótesis: dos versiones de la complementariedad}
\small
Con $D\equiv-\GaR$ (mayor $D$ = cola más adversa):
\begin{itemize}
\item \textbf{H1} (canal bancario): $\beta_1>0$. \quad \textbf{H2} (canal de crecimiento): $\beta_2>0$.
\item \textbf{H3} (complementariedad): la coincidencia amplifica el \textit{spread}.
\end{itemize}
\vspace{2pt}
\begin{block}{Un modelo de intensidad de incumplimiento (Duffie--Singleton, 1999)}
\vspace{-6pt}
\[
\Spread\approx\lambda L,\quad \lambda_{i,t}=\lambda_0\,\JLoss_{i,t-1}^{\beta_1}e^{\beta_2D_{i,t-1}}
\;\Longrightarrow\;\ln\Spread_{i,t}=c+\beta_1\ln\JLoss_{i,t-1}+\beta_2D_{i,t-1}
\]
\vspace{-8pt}
\end{block}
\begin{itemize}
\item En \textbf{puntos básicos} la complementariedad es automática: $\dfrac{\partial^2\Spread}{\partial\JLoss\,\partial D}=\beta_1\beta_2\dfrac{\Spread}{\JLoss}>0$.
\item La versión \textbf{exigente} de H3 es una amplificación \emph{adicional}: $\beta_3>0$ en la forma log-log.
\item $\beta_3=0$ no contradice el marco: es su versión proporcional.
\end{itemize}
\end{frame}

\begin{frame}{Objetivos}
\textbf{Objetivo general.} Evaluar empíricamente el impacto conjunto, estructural y no lineal de la fragilidad bancaria ($\JLoss$) y del riesgo de cola del crecimiento ($\GaR$) sobre el \textit{spread} soberano en economías emergentes.
\vspace{2pt}

\textbf{Objetivos específicos}
\begin{enumerate}\footnotesize
\item Construir y validar $\JLoss$ y $\GaR$ para una muestra amplia de emergentes, de forma reproducible.
\item Efecto marginal de la fragilidad en \textbf{pb y elasticidades}: $\JLoss$ como costo esperado de un rescate sistémico.
\item Efecto marginal del riesgo de cola en \textbf{pb}: $\GaR$ como predictor prospectivo de la solvencia fiscal.
\item Complementariedad mediante $\JLoss\times\GaR$, en \textbf{pb y elasticidades}: amplificación en escenarios de doble vulnerabilidad.
\end{enumerate}
\footnotesize Regresiones lineales de panel con efectos fijos de país y de tiempo; errores Driscoll--Kraay, \textit{cluster} por país y \textit{wild cluster bootstrap}.
\end{frame}

\section{Medición y estrategia empírica}

\begin{frame}{$\JLoss$: fragilidad bancaria sistémica}
\small
Pérdida conjunta esperada del sistema bancario ante un evento sistémico, en \% de la exposición (Chari et al., 2024).
\begin{enumerate}
\item \textbf{PD individual} tipo Merton--KMV: punto de incumplimiento $D^*=L_{ST}+\tfrac12L_{LT}$, $EDF=\Phi(-DD)$.
\item \textbf{Factor sistémico} país (Vasicek): $p_i(V)=\Phi\!\left(\frac{\Phi^{-1}(p_i)-\rho_iV}{\sqrt{1-\rho_i^2}}\right)$.
\item \textbf{Agregación por punto de silla} de la distribución de pérdidas; Gauss--Hermite (7 nodos).
\item \textbf{Contribuciones marginales}: $\JLoss=\dfrac{EL+UL}{\sum EAD}\times100$, LGD $=45\%$.
\end{enumerate}
\vspace{2pt}
\begin{itemize}
\item 113 bancos cotizados, un único protocolo de extracción Bloomberg; reimplementado en Python y validado contra el MATLAB original.
\item La cota de la malla es activa en el 98\% de los casos $\Rightarrow$ $\JLoss$ es \textbf{ordinal}: el logaritmo es invariante a su escala.
\end{itemize}
\end{frame}

\begin{frame}{$\GaR$: riesgo de cola del crecimiento}
\begin{columns}[c]
\column{0.5\textwidth}
\small
\begin{itemize}
\item FCI tipo CISS: bloques accionario, de tasas y cambiario (replica CEMLA, corr.\ 0,9995).
\item Regresión cuantílica de panel, $h=1$:\\
{\footnotesize $Q(\Delta y_{i,t+1}\mid\cdot;\tau)=\alpha+\beta_1\Delta y_{i,t}+\beta_2FCI^{\perp VIX}_{i,t}+\beta_3VIX_t+\mu_i$}
\item $\GaR\equiv$ cuantil 5\%; 39 cuantiles reordenados.
\item \textbf{Pseudo--tiempo real}: ventana expansiva, sin sesgo de anticipación.
\item ¿Circular con el EMBI? El componente tipo \textit{spread} aporta 0,07 al FCI; sin él, el $\GaR$ correlaciona 0,985 con el oficial.
\end{itemize}
\column{0.5\textwidth}
\centering
\includegraphics[width=\textwidth]{fig_skewt_chile_embiext.pdf}\\
{\scriptsize Chile: trimestre benigno vs.\ 2020Q2 ($\GaR=-13{,}4$ pp).}
\end{columns}
\end{frame}

\begin{frame}{Especificación principal}
\begin{block}{Log-log rezagada (Chari et al., 2024)}
\vspace{-6pt}
\[
\ln\EMBI_{i,t}=\alpha_i+\delta_t+\beta_1\ln\JLoss_{i,t-1}+\beta_2D_{i,t-1}+\beta_3(\ln\JLoss_{i,t-1}\times D_{i,t-1})+\omega'X_{i,t}+\varepsilon_{i,t}
\]
\vspace{-8pt}
\end{block}
\footnotesize
\begin{itemize}
\item \textbf{Rezagos}: el \textit{spread} afecta a sus determinantes (bonos en balance, fondeo, FCI); $t-1$ los hace predeterminados.
\item \textbf{Logaritmos}: asimetría del EMBI 1,20 y de $\JLoss$ 2,79 en niveles; elasticidades comparables entre países; $\JLoss$ ordinal.
\item EF de país y de tiempo (absorben VIX, Tesoro, \textit{high yield}); errores Driscoll--Kraay.
\item Sin controles en la principal: deuda, balance fiscal, inflación\ldots\ son plausiblemente \textbf{canales} (mal control). Se agregan como robustez.
\item Batería: 4 modelos anidados $\times$ 3 estructuras de EF $\times$ 3 muestras = 36 regresiones.
\item Vector de crisis: \textit{Backstop} (GFC, COVID) vs.\ \textit{EMstress} (2015Q3--2016Q1) como test de falsación.
\end{itemize}
\end{frame}

\begin{frame}{Resultado 2: ¿efecto o persistencia?}
\small
\begin{block}{La retroalimentación existe y es fuerte}
$\ln\JLoss_t$ sobre $\ln\EMBI_{t-1}$: elasticidad $\mathbf{0{,}17}$ ($p=0{,}009$), mayor que la dirección que se busca medir. Es el bucle de Farhi--Tirole visto desde el otro extremo.
\end{block}
\begin{block}{La dinámica absorbe el efecto de la fragilidad}
\begin{itemize}
\item Persistencia del \textit{spread}: $\rho=0{,}94$ trimestral.
\item Con $\ln\EMBI_{t-1}$: $\beta_1$ de corto plazo $0{,}005$ ($p=0{,}76$); largo plazo $0{,}078$ ($p=0{,}72$). Sesgo de Nickell $\sim1/T$, $T\approx88$.
\item Con la fragilidad adelantada: $\JLoss_{t-1}$ $0{,}058$ ($p=0{,}18$) y $\JLoss_{t+1}$ $0{,}051$ ($p=0{,}23$): un componente persistente común.
\end{itemize}
\end{block}
\end{frame}

\section{Discusión y conclusiones}

\begin{frame}{Síntesis: qué está establecido y qué no}
\begin{columns}[T]
\column{0.48\textwidth}
\begin{block}{Establecido}
\small
\begin{itemize}
\item Fragilidad y riesgo de cola, medidos un trimestre antes, se \textbf{asocian} con un \textit{spread} mayor.
\item La complementariedad \textbf{proporcional} es compatible con los datos.
\item Robusto a Argentina, a la medida de cola y al empalme del EMBI.
\end{itemize}
\end{block}
\column{0.48\textwidth}
\begin{alertblock}{No establecido}
\small
\begin{itemize}
\item Que la fragilidad \textbf{cause} el \textit{spread}; significancia con 13 \textit{clusters}.
\item Una amplificación \textbf{adicional} ($\beta_3>0$) en el promedio del panel.
\item Asimetrías por régimen de crisis: no resisten \textit{cluster} ni permutación.
\end{itemize}
\end{alertblock}
\begin{exampleblock}{Condicional}
\small Amplificación en el núcleo de financiamiento externo, fuera de crisis con respaldo.
\end{exampleblock}
\end{columns}
\end{frame}

\begin{frame}{Interpretación e implicancias}
\begin{itemize}
\small
\item En promedio, la amplificación es \textbf{proporcional}: ambos riesgos escalan la misma intensidad de incumplimiento.
\item No hay una simultaneidad que opere \textbf{siempre y en todas las economías}: aparece cuando no hay respaldo externo y donde la deuda la valoriza un inversionista extranjero.
\item El obstáculo para la causalidad es que la relación \textbf{opera en ambas direcciones}: el mismo \textit{doom loop} que motiva la hipótesis.
\end{itemize}
\vspace{2pt}
\begin{block}{Para la política}
\small Fragilidad bancaria y riesgo de cola son \textbf{indicadores adelantados} del costo de financiamiento soberano. Si la evidencia condicional se confirma, más que una regulación uniforme convendría \textbf{caracterizar los casos} de doble vulnerabilidad: economías de financiamiento externo, en episodios sin prestamista de última instancia.
\end{block}
\end{frame}

\begin{frame}{Limitaciones}
\begin{enumerate}
\item Trece economías (catorce con Argentina): la inferencia robusta no rechaza la ausencia de efectos.
\item \textit{Spread} y fragilidad persistentes y retroalimentados: ni rezagos, ni dinámica, ni IV separan la dirección.
\item Heterogeneidad por economía con grupo definido \textit{ex post}; en logs no resiste controles.
\item Forma funcional no del todo determinada (Box--Cox intermedio).
\item $\JLoss$ ordinal por la calibración; sensible a la malla de pérdidas.
\item Controles anuales interpolados, plausiblemente mediadores; faltan calificación soberana y PIB per cápita.
\item Cobertura desigual del EMBI (Filipinas, Indonesia, Sudáfrica).
\end{enumerate}
\end{frame}

\begin{frame}{Qué haría falta para ir más allá}
\begin{itemize}
\item \textbf{Más economías}: completar el EMBI de Indonesia, Filipinas, Sudáfrica e India; homologar el $\GaR$ de Argentina para incorporarla a la muestra principal.
\item \textbf{Variación exógena específicamente bancaria}: cambios regulatorios, choques de fondeo idiosincrásicos a bancos grandes.
\item \textbf{Mayor frecuencia}: datos diarios/semanales de precios bancarios y \textit{spreads} para separar la dirección de la retroalimentación.
\item \textbf{Participación extranjera} en la deuda soberana como moderador continuo.
\item \textbf{Controles trimestrales} y calificación soberana histórica para completar la réplica de Chari et al.
\item \textit{Pregunta abierta}: ¿amplifica la concentración bancaria? No identificada con tres medidas de HHI.
\end{itemize}
\end{frame}

\begin{frame}{Conclusión}
\begin{itemize}
\item Se articulan el nexo banca--soberano y el \textit{growth-at-risk}, y se construyen ambas métricas para trece emergentes (113 bancos, $\GaR$ en pseudo--tiempo real).
\item Se replica la asociación fragilidad--\textit{spread} y se documenta una análoga para el riesgo de cola.
\item Sometidas a las pruebas que exige el panel, no resisten como efectos causales, y en promedio la complementariedad no va más allá de la proporcional.
\item La amplificación aparece en \textbf{momentos y economías específicas}: evidencia condicional que delimita dónde buscarla.
\end{itemize}
\vspace{2pt}
\begin{block}{El aporte}
Delimitar con precisión qué permite afirmar este diseño y qué no, y dejar una infraestructura de datos reproducible (\texttt{1\_Codigo/Panel/bbg/}) sobre la que un diseño con más países, más frecuencia o variación exógena pueda avanzar.
\end{block}
\end{frame}
