\section{Motivación y pregunta}

\begin{frame}{Dos pilares del \textit{spread} soberano emergente}
\begin{columns}[T]
\column{0.48\textwidth}
\begin{block}{Fundamentos domésticos}
\small
\begin{itemize}
\item Edwards (1984); Hilscher y Nosbusch (2010); Uribe y Yue (2006).
\item \textbf{Proxies:} fragilidad bancaria ($\JLoss$) y riesgo de cola del crecimiento ($\GaR$).
\end{itemize}
\end{block}
\column{0.48\textwidth}
\begin{block}{Factores globales}
\small
\begin{itemize}
\item Calvo et al.\ (1996); Longstaff et al.\ (2011); Miranda-Agrippino y Rey (2022).
\item \textbf{Proxy:} efectos fijos de tiempo (absorben VIX, Tesoro, \textit{high yield}).
\end{itemize}
\end{block}
\end{columns}
\vspace{6pt}
\begin{alertblock}{La brecha}
Chari et al.\ (2024) cruzan la fragilidad bancaria con factores \textbf{globales}. Nadie la ha cruzado con el riesgo de cola \textbf{doméstico} del crecimiento.
\end{alertblock}
\end{frame}

\begin{frame}{Pregunta e hipótesis}
\begin{block}{Pregunta}
¿Penaliza el mercado de forma \textbf{no lineal} la coincidencia de fragilidad bancaria y una cola adversa del crecimiento?
\end{block}
\vspace{4pt}
\begin{itemize}
\item \textbf{H1}\quad La fragilidad bancaria eleva el \textit{spread}.
\item \textbf{H2}\quad El riesgo de cola eleva el \textit{spread}.
\item \textbf{H3}\quad Su coincidencia lo \textbf{amplifica}.
\end{itemize}
\vspace{6pt}
\footnotesize Matiz clave: si ambos escalan la misma probabilidad de incumplimiento, en puntos básicos ya se potencian (\textbf{versión proporcional}). La versión exigente de H3 es una amplificación \textbf{adicional}.
\end{frame}

\begin{frame}{Objetivos}
\textbf{Objetivo general.} Evaluar el impacto conjunto, estructural y no lineal de la fragilidad bancaria ($\JLoss$) y del riesgo de cola del crecimiento ($\GaR$) sobre el \textit{spread} soberano emergente.
\vspace{6pt}

\textbf{Objetivos específicos}
\begin{enumerate}\small
\item Construir y validar las series de $\JLoss$ y $\GaR$, de forma reproducible.
\item Efecto de la fragilidad, en \textbf{puntos básicos y elasticidades}.
\item Efecto del riesgo de cola, en \textbf{puntos básicos}.
\item Complementariedad ($\JLoss\times\GaR$), en \textbf{puntos básicos y elasticidades}.
\end{enumerate}
\end{frame}

\begin{frame}{Qué aporta esta tesis}
\small
\begin{block}{Las métricas no son mías; su construcción y su cruce, sí}
\begin{itemize}
\item $\JLoss$ es la métrica de \textbf{Chari et al.\ (2024)}; el $\GaR$, la de \textbf{Adrian, Boyarchenko y Giannone (2019)}, con la plataforma de CEMLA (Ossandón Busch et al., 2022).
\item \textbf{Lo propio:} reimplementar $\JLoss$ en Python (validada contra el código original) con 113 bancos extraídos con un solo protocolo de Bloomberg, y estimar el $\GaR$ en pseudo--tiempo real para todo el panel.
\end{itemize}
\end{block}
\begin{enumerate}
\item \textbf{Conceptual:} une el nexo banca--soberano y el \textit{growth-at-risk} en un solo mecanismo.
\item \textbf{Empírico:} primer cruce de la fragilidad bancaria con el riesgo de cola \textbf{doméstico}; réplica independiente de Chari et al.; delimita \textbf{cuándo y dónde} aparece la amplificación.
\item \textbf{Metodológico:} \textit{pipeline} reproducible, con cada cifra trazable al código.
\end{enumerate}
\end{frame}

\section{Medición y estrategia}

\begin{frame}{Construcción de $\JLoss$: fragilidad bancaria sistémica}
\small
Pérdida conjunta esperada del sistema bancario ante un evento sistémico, en \% de la exposición (Chari et al., 2024).
\begin{enumerate}
\item \textbf{Probabilidad de incumplimiento de cada banco} (Merton--KMV): $D^*=L_{ST}+\tfrac12L_{LT}$,\quad $EDF=\Phi(-DD)$.
\item \textbf{Factor sistémico} país (Vasicek): $p_i(V)=\Phi\!\left(\frac{\Phi^{-1}(p_i)-\rho_iV}{\sqrt{1-\rho_i^2}}\right)$.
\item \textbf{Agregación por punto de silla} de la distribución de pérdidas del sistema.
\item \textbf{Métrica}: $\JLoss=\dfrac{EL+UL}{\sum_j EAD_j}\times100$.
\end{enumerate}
\vspace{2pt}
{\scriptsize \textbf{Donde:} $D^*$ punto de incumplimiento; $L_{ST}$, $L_{LT}$ pasivos de corto y largo plazo; $DD$ distancia al incumplimiento; $EDF$ frecuencia esperada de incumplimiento; $\Phi$ distribución normal estándar; $p_i$ PD del banco $i$; $V$ factor sistémico (retorno bursátil del país); $\rho_i$ correlación del banco con $V$; $EL$ pérdida esperada; $UL$ pérdida inesperada al 99\%; $EAD$ exposición; LGD $=45\%$. La cota de la malla es activa en el 98\% de los casos $\Rightarrow$ $\JLoss$ es \textbf{ordinal} y entra en logaritmos.\par}
\end{frame}

\begin{frame}{Construcción del $\GaR$: riesgo de cola del crecimiento}
\begin{columns}[c]
\column{0.52\textwidth}
\small
\begin{enumerate}
\item \textbf{Índice de condiciones financieras} (FCI tipo CISS): tensión accionaria, de tasas y cambiaria.
\item \textbf{Regresión cuantílica de panel}, un trimestre adelante:\\
{\footnotesize $Q(\Delta y_{i,t+1}\mid\cdot;\tau)=\alpha+\beta_1\Delta y_{i,t}+\beta_2FCI^{\perp}_{i,t}+\beta_3VIX_t+\mu_i$}
\item $\GaR_{i,t}$ = cuantil $\tau=5\%$; ventana expansiva (pseudo--tiempo real).
\end{enumerate}
\vspace{2pt}
{\scriptsize \textbf{Donde:} $Q(\cdot;\tau)$ cuantil $\tau$ del crecimiento; $\Delta y$ crecimiento del PIB; $FCI^{\perp}$ FCI sin el componente del VIX; $\mu_i$ efecto fijo de país. En la tesis $D\equiv-\GaR$: mayor $D$, cola más adversa. No es circular con el EMBI (corr.\ 0,985 sin el componente tipo \textit{spread}).\par}
\column{0.46\textwidth}
\centering
\includegraphics[width=\textwidth]{fig_skewt_chile_embiext.pdf}\\
{\scriptsize Chile: trimestre benigno vs.\ 2020Q2 ($\GaR=-13{,}4$ pp).}
\end{columns}
\end{frame}

\begin{frame}{Especificación principal y batería}
\begin{block}{Log-log rezagada}
\vspace{-6pt}
\[
\ln\EMBI_{i,t}=\alpha_i+\delta_t+\beta_1\ln\JLoss_{i,t-1}+\beta_2D_{i,t-1}+\beta_3\,(\ln\JLoss\times D)_{i,t-1}+\varepsilon_{i,t}
\]
\vspace{-8pt}
\end{block}
{\scriptsize \textbf{Donde:} $i$ país, $t$ trimestre; $\ln\EMBI$ logaritmo del \textit{spread} soberano; $\alpha_i$ efecto fijo de país; $\delta_t$ efecto fijo de tiempo; $D=-\GaR$ riesgo de cola (pp); $\beta_1$ elasticidad a la fragilidad; $\beta_2$ semielasticidad al riesgo de cola; $\beta_3$ interacción (amplificación adicional si $>0$); $\varepsilon$ error.\par}
\vspace{4pt}
\begin{columns}[T]
\column{0.5\textwidth}
\small
\textbf{Batería: 4 modelos anidados}
\begin{itemize}
\item M1: solo fragilidad \quad M2: solo cola
\item M3: ambas \quad M4: ambas + interacción
\end{itemize}
\column{0.48\textwidth}
\small
\textbf{× 3 efectos fijos × 3 muestras}
\begin{itemize}
\item EF: tiempo (T), país (P), ambos (PT)
\item Completa, sin crisis, con 6 controles
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}{Especificación con el vector de crisis}
\begin{block}{Interacción con crisis, sin botar trimestres}
\small
\vspace{-6pt}
\[
\begin{aligned}\ln\EMBI_{i,t}={}&\alpha_i+\delta_t+\beta_1\ln\JLoss_{i,t-1}+\beta_2D_{i,t-1}+\beta_3(\ln\JLoss\times D)_{i,t-1}\\&+\beta_4(\ln\JLoss\times D)_{i,t-1}C_t+\beta_5\ln\JLoss_{i,t-1}C_t+\beta_6D_{i,t-1}C_t+\omega'X_{i,t}+\varepsilon_{i,t}\end{aligned}
\]
\vspace{-8pt}
\end{block}
{\scriptsize \textbf{Donde:} $C_t$ indicador de crisis; $\beta_3$ interacción \textbf{fuera} de crisis; $\beta_3+\beta_4$ interacción \textbf{dentro} de la crisis; $\beta_5$, $\beta_6$ cambios de la fragilidad y de la cola en crisis; $X$ seis controles domésticos (deuda, balance fiscal, reservas, cuenta corriente, inflación, tipo de cambio real); $\omega$ sus coeficientes. CM1--CM4: mismos modelos anidados con sus interacciones de crisis.\par}
\vspace{4pt}
\begin{columns}[T]
\column{0.48\textwidth}
\begin{block}{Vector único}
\small $C_t$: 2008Q4--2009Q4, 2015Q3--2016Q1 y 2020Q1--2021Q4.
\end{block}
\column{0.48\textwidth}
\begin{block}{Test de falsación}
\small \textit{Backstop}: GFC y COVID (con Fed y FMI).\\ \textit{EMstress}: 2015--16 (sin respaldo).
\end{block}
\end{columns}
\vspace{2pt}
\centering\footnotesize Si el respaldo oficial apaga el canal: $\beta_3+\beta_4\approx0$ bajo \textit{Backstop} y $>0$ bajo \textit{EMstress}.
\end{frame}

\begin{frame}{Inferencia: cómo mido la incertidumbre}
\small
\begin{itemize}
\item \textbf{Driscoll--Kraay (referencia):} robustos a heterocedasticidad, autocorrelación del \textit{spread} y choques comunes entre países.
\vspace{3pt}
\item \textbf{\textit{Cluster} por país:} robustos a cualquier autocorrelación dentro de cada economía.
\vspace{3pt}
\item \textbf{\textit{Wild cluster bootstrap}} (pesos de Webb, 1\,999 réplicas): con solo 13 países los errores asintóticos son optimistas; un resultado es robusto solo si resiste esta prueba.
\vspace{3pt}
\item \textbf{Regresor generado:} el $\GaR$ es una estimación; se propaga su error con 500 réplicas.
\vspace{3pt}
\item \textbf{Identificación:} canal inverso, dinámica, proyecciones locales, variables instrumentales y permutación.
\end{itemize}
\end{frame}

\begin{frame}{Cómo leer este hallazgo: es condicional}

\small
\begin{itemize}
\item Resiste el \textit{wild bootstrap} ($p=0{,}015$ y $0{,}002$).
\item Aparece en logs y en niveles.
\item Sobrevive a excluir cada país.
\item El contraste va en sentido opuesto.
\end{itemize}
\end{block}

\begin{alertblock}{Lo limita}

\begin{itemize}

\item En logs, los controles lo absorben.
\item Solo 11 países.
\item 2015--16 no pasa la permutación.
\end{itemize}
\end{alertblock}
\end{columns}

Una hipótesis \textbf{bien delimitada} para contrastar con más datos, no un hallazgo robusto.

\section{Conclusiones}

\begin{frame}{Respuesta a la pregunta}
\begin{enumerate}
\item \textbf{Métricas}: $\JLoss$ y $\GaR$ construidas y validadas para 13 emergentes.
\vspace{3pt}
\item \textbf{Fragilidad}: elasticidad 0,10 ($\approx$4,7 pb por unidad); asociación, no causal.
\vspace{3pt}
\item \textbf{Riesgo de cola}: +3,3\% por punto; asociación, no causal.
\vspace{3pt}
\item \textbf{Complementariedad}: proporcional en promedio; adicional solo condicional (+0,034; +1,8 pb).
\end{enumerate}
\vspace{6pt}
\begin{block}{¿Penaliza el mercado la coincidencia de forma no lineal?}
\textbf{No en general. Sí, condicionalmente}: sin respaldo externo y en economías de financiamiento externo.
\end{block}
\end{frame}

\begin{frame}{Hallazgos específicos y su grado de evidencia}
\centering
\footnotesize
\begin{adjustbox}{max width=\textwidth}
\begin{tabular}{llc}
\toprule
Hallazgo & Cifra & Evidencia \\
\midrule
La fragilidad bancaria se asocia con el \textit{spread} & 0,10 ($\approx$4,7 pb por unidad) & establecida \\
El riesgo de cola se asocia con el \textit{spread} & +3,3\% por punto & establecida \\
Se replica a Chari et al.\ (2024) & 0,107--0,164 & establecida \\
Los controles domésticos absorben la fragilidad & $0{,}10\to-0{,}008$ & establecida (canales) \\
El \textit{spread} retroalimenta a la fragilidad & 0,17 & establecida \\
Efecto causal de la fragilidad o de la cola & \textit{wild} $p=0{,}33$ / $0{,}20$ & \textbf{no establecido} \\
Amplificación adicional en promedio & $\beta_3=-0{,}011$ & \textbf{no hay} \\
Interacción negativa bajo \textit{Backstop} & $-0{,}015$ (\textit{cluster} $p=0{,}40$) & no robusta \\
Amplificación en 2015--16 (\textit{EMstress}) & $+0{,}026$; lugar 20 de 51 & no robusta \\
Amplificación en el núcleo, sin crisis & $+0{,}034$; $+1{,}8$ pb & \textbf{condicional} \\
Polonia e India, en sentido opuesto & $-0{,}118$ & condicional \\
Argentina y otras robusteces & sin cambios & robusta \\
\bottomrule
\end{tabular}
\end{adjustbox}
\end{frame}

\begin{frame}{Dónde buscar la amplificación}
\begin{columns}[T]
\column{0.5\textwidth}
\small
\textbf{Cuándo}
\begin{itemize}
\item Fuera de las crisis con respaldo oficial masivo (GFC, COVID).
\item Cuando coinciden fragilidad alta y cola adversa sin un prestamista de última instancia.
\end{itemize}
\vspace{3pt}
\textbf{Dónde}
\begin{itemize}
\item Brasil, Chile, China, Colombia, Filipinas, Indonesia, Malasia, México, Perú, Sudáfrica y Turquía.
\item No en Polonia ni India: deuda local profunda.
\end{itemize}
\column{0.48\textwidth}
\small
\textbf{Por qué}
\begin{itemize}
\item El precio lo fija un inversionista extranjero, sensible al riesgo del rescate.
\end{itemize}
\vspace{3pt}
\textbf{Qué vigilar}
\begin{itemize}
\item $\JLoss$ y $\GaR$ juntos, y la participación extranjera en la deuda.
\end{itemize}
\vspace{3pt}
\textbf{Qué falta para confirmarlo}
\begin{itemize}
\item Un grupo definido \textit{ex ante}, más países y más episodios sin respaldo.
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}{Implicancias}
\begin{columns}[T]
\column{0.48\textwidth}
\begin{block}{Respaldado por los datos}
\small
Monitorear \textbf{juntos} $\JLoss$ y $\GaR$: ambos adelantan un \textit{spread} más alto.
\end{block}
\column{0.48\textwidth}
\begin{exampleblock}{Si se confirma lo condicional}
\small
\textbf{Focalizar} en vez de regular igual a todos: economías de financiamiento externo, en episodios sin respaldo oficial.
\end{exampleblock}
\end{columns}
\vspace{8pt}
\begin{alertblock}{No permite}
Cuantificar cuánto bajaría el \textit{spread} si cae la fragilidad.
\end{alertblock}
\end{frame}

\begin{frame}{Limitaciones y trabajo futuro}
\begin{columns}[T]
\column{0.48\textwidth}
\begin{alertblock}{Limitaciones}
\small
\begin{itemize}
\item 13 países: inferencia frágil.
\item Relación de doble vía y muy persistente.
\item Heterogeneidad definida \textit{ex post}.
\item $\JLoss$ ordinal; controles anuales.
\end{itemize}
\end{alertblock}
\column{0.48\textwidth}
\begin{block}{Trabajo futuro}
\small
\begin{itemize}
\item Más países (incorporar Argentina).
\item Participación extranjera medida \textit{ex ante}.
\item Variación exógena bancaria.
\item Datos de mayor frecuencia.
\end{itemize}
\end{block}
\end{columns}
\end{frame}

\begin{frame}{Conclusión}
\vspace{6pt}
\begin{enumerate}
\item La fragilidad bancaria y el riesgo de cola \textbf{se asocian} con el \textit{spread} soberano.
\vspace{8pt}
\item No es causal: el \textit{spread} también \textbf{retroalimenta} a la fragilidad.
\vspace{8pt}
\item La amplificación \textbf{no es universal}: aparece fuera de crisis con respaldo y en economías de financiamiento externo.
\end{enumerate}
\vspace{10pt}
\centering
\textit{El aporte es saber dónde buscarla y qué haría falta para confirmarla.}
\end{frame}
