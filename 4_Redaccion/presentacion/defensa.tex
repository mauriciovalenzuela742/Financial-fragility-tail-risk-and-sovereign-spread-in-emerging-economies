\section{Motivación y pregunta}

\begin{frame}{Dos pilares del \textit{spread} soberano emergente}
\begin{columns}[T]
\column{0.48\textwidth}
\begin{block}{Fundamentos domésticos}
\small
\begin{itemize}
\item Edwards (1984); Hilscher y Nosbusch (2010); Uribe y Yue (2006).
\item \textbf{Mis proxies:} fragilidad bancaria ($\JLoss$) y riesgo de cola del crecimiento ($\GaR$).
\end{itemize}
\end{block}
\column{0.48\textwidth}
\begin{block}{Factores globales}
\small
\begin{itemize}
\item Calvo et al.\ (1996); Longstaff et al.\ (2011); Miranda-Agrippino y Rey (2022).
\item \textbf{Mi proxy:} efectos fijos de tiempo (VIX, Tesoro, \textit{high yield}).
\end{itemize}
\end{block}
\end{columns}
\vspace{6pt}
\begin{alertblock}{La brecha}
Chari et al.\ (2024) cruzan la fragilidad bancaria con factores \textbf{globales}. Nadie la ha cruzado con el riesgo de cola \textbf{doméstico} del crecimiento.
\end{alertblock}
\end{frame}

\begin{frame}{Pregunta e hipótesis}
\begin{block}{Pregunta}
¿Penaliza el mercado de forma \textbf{no lineal} la coincidencia de fragilidad bancaria y una cola adversa del crecimiento?
\end{block}
\vspace{4pt}
\begin{itemize}
\item \textbf{H1}\quad La fragilidad bancaria eleva el \textit{spread}.
\item \textbf{H2}\quad El riesgo de cola eleva el \textit{spread}.
\item \textbf{H3}\quad Su coincidencia lo \textbf{amplifica}.
\end{itemize}
\vspace{6pt}
\footnotesize Matiz clave: si ambos escalan la misma probabilidad de incumplimiento, en puntos básicos ya se potencian (\textbf{versión proporcional}). La versión exigente de H3 es una amplificación \textbf{adicional}.
\end{frame}

\begin{frame}{Qué aporta esta tesis}
\begin{enumerate}
\item \textbf{Conceptual.} Une dos literaturas separadas ---nexo banca--soberano y \textit{growth-at-risk}--- en un solo mecanismo.
\vspace{6pt}
\item \textbf{Empírico.} Construye ambas métricas para 13 emergentes, replica a Chari et al.\ (2024) con datos propios y delimita \textbf{cuándo} y \textbf{en qué economías} aparece la amplificación.
\vspace{6pt}
\item \textbf{Metodológico.} \textit{Pipeline} reproducible: 113 bancos desde Bloomberg y $\GaR$ en pseudo--tiempo real.
\end{enumerate}
\vspace{8pt}
\begin{alertblock}{Con un límite explícito}
Los hallazgos son \textbf{asociaciones}, no efectos causales, y la amplificación es \textbf{condicional}.
\end{alertblock}
\end{frame}

\section{Medición y estrategia}

\begin{frame}{Dos métricas construidas desde cero}
\begin{columns}[T]
\column{0.48\textwidth}
\begin{block}{$\JLoss$: fragilidad bancaria}
\small
\begin{itemize}
\item Pérdida conjunta esperada del sistema bancario.
\item PD tipo Merton $\to$ agregación por punto de silla.
\item 113 bancos; validada contra el código original.
\item Ordinal $\Rightarrow$ entra en logaritmos.
\end{itemize}
\end{block}
\column{0.48\textwidth}
\begin{block}{$\GaR$: riesgo de cola}
\small
\begin{itemize}
\item Cuantil 5\% del crecimiento futuro.
\item Regresión cuantílica de panel sobre condiciones financieras.
\item Pseudo--tiempo real: sin mirar el futuro.
\item No es circular con el EMBI (corr.\ 0,985 sin el componente tipo \textit{spread}).
\end{itemize}
\end{block}
\end{columns}
\end{frame}

\begin{frame}{La especificación}
\begin{block}{Log-log rezagada}
\vspace{-6pt}
\[
\ln\EMBI_{i,t}=\alpha_i+\delta_t+\beta_1\ln\JLoss_{i,t-1}+\beta_2D_{i,t-1}+\beta_3\,(\ln\JLoss\times D)_{i,t-1}+\varepsilon_{i,t}
\]
\vspace{-8pt}
\end{block}
{\small con $D\equiv-\GaR$: mayor $D$, cola más adversa.}
\vspace{4pt}
\begin{itemize}
\item \textbf{Rezagos}: frente a la causalidad inversa.
\item \textbf{Logaritmos}: elasticidades; series muy asimétricas.
\item \textbf{Efectos fijos} de país y de tiempo.
\item \textbf{Errores}: Driscoll--Kraay y \textit{wild cluster bootstrap} (pocos países).
\item Todo se repite en \textbf{puntos básicos} (niveles).
\end{itemize}
\end{frame}

\section{Resultados}

\begin{frame}{1 · Ambos riesgos se asocian con el \textit{spread}}
\begin{columns}[T]
\column{0.48\textwidth}
\cifra{0,10}{elasticidad a la fragilidad bancaria\\ (duplicar $\JLoss$ $\approx$ +7\%; $\approx$4,7 pb por unidad)}
\column{0.48\textwidth}
\cifra{+3,3\,\%}{de \textit{spread} por cada punto\\ de riesgo de cola}
\end{columns}
\vspace{10pt}
\begin{itemize}
\item Replica a Chari et al.\ (2024): elasticidad 0,11--0,16 con sus controles.
\item Con controles fiscales y cambiarios la fragilidad se diluye: son plausiblemente \textbf{canales}.
\end{itemize}
\end{frame}

\item Resiste el \textit{wild bootstrap} ($p=0{,}015$ y $0{,}002$).

\section{Conclusiones}

\begin{frame}{Respuesta a la pregunta}
\begin{enumerate}
\item \textbf{Métricas}: $\JLoss$ y $\GaR$ construidas y validadas para 13 emergentes.
\vspace{4pt}
\item \textbf{Fragilidad}: se asocia con el \textit{spread} (0,10; $\approx$4,7 pb), no de forma causal.
\vspace{4pt}
\item \textbf{Riesgo de cola}: se asocia con el \textit{spread} (+3,3\%), no de forma causal.
\vspace{4pt}
\item \textbf{Complementariedad}: proporcional en promedio; amplificación adicional solo \textbf{condicional}.
\end{enumerate}
\vspace{6pt}
\begin{block}{¿Penaliza el mercado la coincidencia de forma no lineal?}
\textbf{No en general. Sí, condicionalmente}: sin respaldo externo y en economías de financiamiento externo.
\end{block}
\end{frame}

\begin{frame}{El aporte, en una lámina}
\begin{itemize}
\item \textbf{Un mecanismo} que une la fragilidad bancaria y el riesgo de cola del crecimiento.
\vspace{4pt}
\item \textbf{Dos métricas propias} y una réplica independiente de Chari et al.\ (2024).
\vspace{4pt}
\item \textbf{Una delimitación precisa}: qué se puede afirmar, qué no, y dónde buscar la amplificación.
\vspace{4pt}
\item \textbf{Una infraestructura reproducible} para ir más allá.
\end{itemize}
\vspace{8pt}
\begin{alertblock}{Lo que no afirma}
Ni causalidad ni una amplificación universal.
\end{alertblock}
\end{frame}

\begin{frame}{Implicancias}
\begin{columns}[T]
\column{0.48\textwidth}
\begin{block}{Respaldado por los datos}
\small
Monitorear \textbf{juntos} $\JLoss$ y $\GaR$: ambos adelantan un \textit{spread} más alto.
\end{block}
\column{0.48\textwidth}
\begin{exampleblock}{Si se confirma lo condicional}
\small
\textbf{Focalizar} en vez de regular igual a todos: economías de financiamiento externo, en episodios sin respaldo oficial.
\end{exampleblock}
\end{columns}
\vspace{8pt}
\begin{alertblock}{No permite}
Cuantificar cuánto bajaría el \textit{spread} si cae la fragilidad.
\end{alertblock}
\end{frame}

\begin{frame}{Trabajo futuro}
\begin{itemize}
\item \textbf{Más países}: completar el EMBI de Asia y Sudáfrica; incorporar Argentina.
\vspace{4pt}
\item \textbf{Moderador medido ex ante}: participación extranjera en la deuda.
\vspace{4pt}
\item \textbf{Variación exógena bancaria}: cambios regulatorios, choques de fondeo.
\vspace{4pt}
\item \textbf{Mayor frecuencia}: datos diarios o semanales.
\end{itemize}
\end{frame}

\begin{frame}{Conclusión}
\vspace{6pt}
\begin{enumerate}
\item La fragilidad bancaria y el riesgo de cola \textbf{se asocian} con el \textit{spread} soberano.
\vspace{8pt}
\item No es causal: el \textit{spread} también \textbf{retroalimenta} a la fragilidad.
\vspace{8pt}
\item La amplificación \textbf{no es universal}: aparece en momentos y economías específicas.
\end{enumerate}
\vspace{10pt}
\centering
\textit{El aporte es saber dónde buscarla y qué haría falta para confirmarla.}
\end{frame}

\begin{frame}{Respaldo: construcción de $\JLoss$}
\small
Pérdida conjunta esperada del sistema bancario ante un evento sistémico, en \% de la exposición (Chari et al., 2024).
\begin{enumerate}
\item \textbf{PD individual} tipo Merton--KMV: $D^*=L_{ST}+\tfrac12L_{LT}$, $EDF=\Phi(-DD)$.
\item \textbf{Factor sistémico} país (Vasicek): $p_i(V)=\Phi\!\left(\frac{\Phi^{-1}(p_i)-\rho_iV}{\sqrt{1-\rho_i^2}}\right)$.
\item \textbf{Agregación por punto de silla}; Gauss--Hermite (7 nodos).
\item \textbf{Contribuciones marginales}: $\JLoss=\dfrac{EL+UL}{\sum EAD}\times100$, LGD $=45\%$.
\end{enumerate}
\vspace{2pt}
La cota de la malla es activa en el 98\% de los casos $\Rightarrow$ $\JLoss$ es \textbf{ordinal}.
\end{frame}

\begin{frame}{Respaldo: limitaciones}
\small
\begin{enumerate}
\item Trece economías (catorce con Argentina): la inferencia robusta no rechaza la ausencia de efectos.
\item \textit{Spread} y fragilidad persistentes y retroalimentados: ni rezagos, ni dinámica, ni IV separan la dirección.
\item Heterogeneidad por economía con grupo definido \textit{ex post}; en logs no resiste controles.
\item Forma funcional no del todo determinada (Box--Cox intermedio).
\item $\JLoss$ ordinal; sensible a la malla de pérdidas.
\item Controles anuales interpolados; faltan calificación soberana y PIB per cápita.
\item Cobertura desigual del EMBI (Filipinas, Indonesia, Sudáfrica).
\end{enumerate}
\end{frame}
